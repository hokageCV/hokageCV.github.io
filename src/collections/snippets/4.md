const button = document.querySelector('<the-selectors>');

// Trigger the hover events
if (button) {
  button.dispatchEvent(new MouseEvent('mouseover', { bubbles: true }));
  button.dispatchEvent(new MouseEvent('mouseenter', { bubbles: true }));
  console.log('Hover triggered successfully!');
} else {
  console.error('Element not found. Make sure you are inside the correct iframe.');
}
